    24→# 1. Platform Philosophy
    25→
    26→## Core Principle: Candidate Empowerment, Not Disqualification
    27→
    28→Our platform is designed to **open doors** for job seekers, not close them.
    29→
    30→### The Kenyan Reality:
    31→- A graduate may work multiple jobs for survival
    32→- A BCom graduate might be: Cashier (Today) → Teller (Next Year) → Auditor (Following Year)
    33→- Job seekers have **multiple professional identities**, not one fixed career path
    34→
    35→### What We Do:
    36→- Acknowledge all work experiences as valid
    37→- Present jobs that match **ANY** of the candidate's experiences
    38→- Allow candidates to choose up to **3 career trajectories** to focus on
    39→- Never disqualify candidates based on missing constraints
    40→- Let candidates decide if they meet administrative requirements (e.g., driving license, age)
    41→- Rank jobs so best matches appear first, but all matching jobs are visible
    42→- Show location as information, never as a filter
    43→
    44→### What We Don't Do:
    45→- Penalize candidates for having unknown employers
    46→- Lock candidates into a single career path
    47→- Use hard filters that reject candidates
    48→- Extract or rely on ambiguous data like company sector from CVs
    49→- Match on administrative requirements (license, age, nationality, languages)
    50→- Force candidates into rigid career categories
    51→- Filter jobs by location (candidates decide if they want to relocate)

    55→# 2. System Architecture
    56→
    57→## 2.1 The "Extract Once, Compute Many" Approach
    58→
    59→**Key Decision:** LLM fires **ONCE** for CV parsing and **ONLY WHEN NEEDED** for JD parsing. All matching happens in the database/backend using structured data.
    60→
    61→### Why This Approach:
    62→
    63→| Factor | Traditional (LLM Every Time) | Our Approach (Extract Once) |
    64→|--------|------------------------------|------------------------------|
    65→| **Cost** | $0.001 per match | $0.001 per upload + $0 for matches |
    66→| **Speed** | Slow (LLM latency) | Instant (DB query) |
    67→| **Scalability** | Linear cost scaling | Fixed cost per upload |
    68→| **Accuracy** | Inconsistent (LLM varies) | Deterministic (math) |
    69→

   325→# 4. Data Normalization & Enums
   326→
   327→## 4.1 Why Normalization Matters
   328→
   329→Without normalization, the same concept appears in multiple forms:
   330→
   331→| Raw LLM Output | Issue |
   332→|----------------|-------|
   333→| "BSc", "B.Sc", "Bachelor of Science", "Bachelors" | 4 variations of "Bachelor" |
   334→| "Admin", "Administration", "Office Admin" | 3 variations of "Administration" |
   335→| "Telco", "Telecom", "Telecommunications" | 3 variations of "Telecommunications" |
   336→
   337→**Normalization** maps all variations to a single canonical value.

   412→## 4.6 Normalization Implementation
   413→
   414→### Word-Boundary Matching (Avoids False Positives)
   415→
   416→```javascript
   417→// lib/normalize.js
   418→import { EDUCATION_LEVEL_ENUMS, FUNCTION_ENUMS, SECTOR_ENUMS } from './enums';
   419→
   420→/**
   421→ * Normalize using word-boundary regex to avoid false matches
   422→ * "hr" matches "hr" but NOT "research" or "architecture"
   423→ */
   424→function normalizeWithWordBoundaries(raw, enumMap, defaultVal = 'Other') {
   425→    if (!raw) return defaultVal;
   426→    const lower = raw.toLowerCase().trim();
   427→    
   428→    for (const [canonical, variations] of Object.entries(enumMap)) {
   429→        for (const variation of variations) {
   430→            // Use word boundaries to avoid substring false positives
   431→            const regex = new RegExp(`\\b${variation.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')}\\b`, 'i');
   432→            if (regex.test(lower)) {
   433→                return canonical;
   434→            }
   435→        }
   436→    }
   437→    return defaultVal;
   438→}
   439→
   440→export function normalizeEducationLevel(raw) {
   441→    return normalizeWithWordBoundaries(raw, EDUCATION_LEVEL_ENUMS, 'None');
   442→}
   443→
   444→export function normalizeFunction(raw) {
   445→    return normalizeWithWordBoundaries(raw, FUNCTION_ENUMS, 'Other');
   446→}
   447→
   448→export function normalizeSector(raw) {
   449→    return normalizeWithWordBoundaries(raw, SECTOR_ENUMS, 'Other');
   450→}
   451→
   452→export function normalizeCandidate(rawJson) {
   453→    if (!rawJson) return null;
   454→    
   455→    // Education is now an array
   456→    if (rawJson.education && Array.isArray(rawJson.education)) {
   457→        for (const edu of rawJson.education) {
   458→            edu.level = normalizeEducationLevel(edu.level);
   459→        }
   460→    }
   461→    
   462→    // Normalize work experience clusters
   463→    if (rawJson.work_experience) {
   464→        for (const cluster of rawJson.work_experience) {
   465→            cluster.function = normalizeFunction(cluster.function);
   466→            
   467→            // De-duplicate skills
   468→            if (cluster.skills) {
   469→                cluster.skills = [...new Set(cluster.skills.map(s => s.trim()))];
   470→            }
   471→        }
   472→    }
   473→    
   474→    return rawJson;
   475→}
   476→
   477→export function normalizeJob(rawJson) {
   478→    if (!rawJson) return null;
   479→    
   480→    if (rawJson.job) {
   481→        rawJson.job.sector = normalizeSector(rawJson.job.sector);
   482→        rawJson.job.job_function = normalizeFunction(rawJson.job.job_function);
   483→        if (rawJson.job.required_education) {
   484→            rawJson.job.required_education.level = normalizeEducationLevel(
   485→                rawJson.job.required_education.level
   486→            );
   487→        }
   488→    }
   489→    
   490→    return rawJson;
   491→}
   492→```

   494→## 4.7 Field Relatedness Mapping
   495→
   496→```javascript
   497→// lib/field-mapping.js
   498→export const RELATED_FIELDS = {
   499→    'business': ['commerce', 'business administration', 'marketing', 'finance', 'accounting', 'management'],
   500→    'computing': ['computer science', 'it', 'information technology', 'software', 'data science', 'programming'],
   501→    'engineering': ['mechanical engineering', 'civil engineering', 'electrical engineering', 'automotive engineering', 'structural engineering'],
   502→    'health': ['medicine', 'nursing', 'clinical', 'pharmacy', 'public health', 'healthcare'],
   503→    'education': ['teaching', 'training', 'curriculum', 'pedagogy', 'instruction'],
   504→    'economics': ['econometrics', 'finance', 'statistics', 'actuarial science'],
   505→    'law': ['legal', 'jurisprudence', 'advocacy', 'constitutional law'],
   506→    'arts': ['humanities', 'social sciences', 'literature', 'history', 'sociology']
   507→};
   508→
   509→export function isFieldRelated(candidateField, jobField) {
   510→    if (!candidateField || !jobField) return false;
   511→    const c = candidateField.toLowerCase().trim();
   512→    const j = jobField.toLowerCase().trim();
   513→    
   514→    // Direct match
   515→    if (c === j) return true;
   516→    
   517→    // Check if one contains the other (but avoid false positives)
   518→    if (c.includes(j) || j.includes(c)) {
   519→        // Only return true if the shorter string is at least 3 characters
   520→        // and the match is at a word boundary
   521→        const shorter = c.length <= j.length ? c : j;
   522→        const longer = c.length > j.length ? c : j;
   523→        if (shorter.length >= 3 && longer.includes(shorter)) {
   524→            // Check if it's a word boundary match
   525→            const regex = new RegExp(`\\b${shorter}\\b`, 'i');
   526→            if (regex.test(longer)) return true;
   527→        }
   528→    }
   529→    
   530→    // Check related fields map
   531→    for (const [key, related] of Object.entries(RELATED_FIELDS)) {
   532→        const isCandidateRelated = related.some(r => c.includes(r) || r.includes(c));
   533→        const isJobRelated = related.some(r => j.includes(r) || r.includes(j));
   534→        if (isCandidateRelated && isJobRelated) return true;
   535→    }
   536→    
   537→    return false;
   538→}

   650→## 5.4 LLM Error Handling & Retry Logic
   651→
   652→```javascript
   653→// lib/llm.js
   654→import { GoogleGenerativeAI } from '@google/generative-ai';
   655→
   656→const genAI = new GoogleGenerativeAI(process.env.GEMINI_API_KEY);
   657→const model = genAI.getGenerativeModel({ model: 'gemini-1.5-flash' });
   658→
   659→async function parseWithRetry(prompt, maxRetries = 3) {
   660→    let lastError = null;
   661→    let attempt = 0;
   662→    
   663→    while (attempt < maxRetries) {
   664→        try {
   665→            const result = await model.generateContent(prompt);
   666→            const response = await result.response;
   667→            const jsonText = response.text();
   668→            
   669→            // Try to parse JSON
   670→            const parsed = JSON.parse(jsonText);
   671→            
   672→            // Basic validation
   673→            if (!parsed.profile?.full_name && attempt < 2) {
   674→                throw new Error('Missing required field: full_name');
   675→            }
   676→            
   677→            return parsed;
   678→            
   679→        } catch (error) {
   680→            lastError = error;
   681→            attempt++;
   682→            console.warn(`LLM attempt ${attempt} failed:`, error.message);
   683→            
   684→            if (attempt < maxRetries) {
   685→                await sleep(1000 * attempt); // Exponential backoff
   686→            }
   687→        }
   688→    }
   689→    
   690→    // All retries failed - log for manual review
   691→    await logFailedParsing(prompt, lastError);
   692→    
   693→    return {
   694→        _error: true,
   695→        _message: 'Failed to parse. Please review manually.',
   696→        _raw_text: prompt.substring(0, 500) // Truncate for logging
   697→    };
   698→}
   699→
   700→function sleep(ms) {
   701→    return new Promise(resolve => setTimeout(resolve, ms));
   702→}
   703→
   704→async function logFailedParsing(text, error) {
   705→    // Log to database for manual review
   706→    await query(
   707→        `INSERT INTO parse_failures (raw_text, error_message, created_at) 
   708→         VALUES (?, ?, NOW())`,
   709→        [text.substring(0, 10000), error.message]
   710→    );
   711→}
   712→
   713→export async function parseCV(text) {
   714→    const prompt = `...`; // Full prompt from above
   715→    return await parseWithRetry(prompt);
   716→}
   717→
   718→export async function parseJD(text) {
   719→    const prompt = `...`; // Full prompt from above
   720→    return await parseWithRetry(prompt);
   721→}

   726→# 6. Matching Algorithm
   727→
   728→## 6.1 Core Principles
   729→
   730→| Principle | Implementation |
   731→|-----------|----------------|
   732→| **Inclusive** | Never hides jobs based on missing data |
   733→| **Function-Based** | Primary match is on job function |
   734→| **Ranked** | Best matches appear first |
   735→| **Explainable** | Each match shows why it was recommended |
   736→| **User-Controlled** | Only selected trajectories are matched |
   737→| **Scalable** | Uses database indexing for efficient queries |
   738→
   739→## 6.2 Step-by-Step Logic
   740→
   741→### Step 1: Function Match (Gateway Filter)
   742→
   743→```javascript
   744→// lib/matching.js
   745→function isFunctionMatch(candidate, job) {
   746→    // Only check selected clusters
   747→    const activeClusters = candidate.work_experience.filter(c => c.selected);
   748→    
   749→    for (const cluster of activeClusters) {
   750→        if (cluster.function.toLowerCase() === job.job_function.toLowerCase()) {
   751→            return true;
   752→        }
   753→    }
   754→    return false;
   755→}
   756→```
   757→
   758→**Outcome:**
   759→- If **any** selected cluster matches → Show the job
   760→- If **no** selected cluster matches → Hide the job
   761→
   762→### Step 2: Calculate Match Score (Ranking)
   763→
   764→```javascript
   765→function calculateMatchScore(cluster, job, candidate) {
   766→    let score = 0;
   767→    
   768→    // 1. Job Title Match (40 points)
   769→    const jobTitleLower = job.title.toLowerCase();
   770→    const hasExactMatch = cluster.job_titles.some(
   771→        t => t.toLowerCase() === jobTitleLower
   772→    );
   773→    
   774→    if (hasExactMatch) {
   775→        score += 40;
   776→    } else {
   777→        // Partial match - check for keyword overlap
   778→        const jobKeywords = new Set(job.title.toLowerCase().split(' '));
   779→        const titleKeywords = new Set(
   780→            cluster.job_titles.join(' ').toLowerCase().split(' ')
   781→        );
   782→        const overlap = [...jobKeywords].filter(k => titleKeywords.has(k));
   783→        if (overlap.length > 0) {
   784→            score += Math.min(20, overlap.length * 5);
   785→        }
   786→    }
   787→    
   788→    // 2. Skills Match (35 points)
   789→    const matchedSkills = job.required_skills.filter(
   790→        skill => cluster.skills.some(s => s.toLowerCase() === skill.toLowerCase())
   791→    );
   792→    
   793→    if (job.required_skills.length > 0) {
   794→        const skillPercentage = matchedSkills.length / job.required_skills.length;
   795→        score += skillPercentage * 35;
   796→    }
   797→    
   798→    // 3. Education Match (15 points)
   799→    // Use highest education level from array
   800→    const highestLevel = getHighestEducationLevel(candidate.education);
   801→    if (job.required_education.level) {
   802→        const eduScore = compareEducation(
   803→            highestLevel,
   804→            job.required_education.level
   805→        );
   806→        score += eduScore * 15;
   807→        
   808→        // Bonus for field relevance (5 points)
   809→        if (job.required_education.field && 
   810→            job.required_education.field !== "Any") {
   811→            const isRelated = isFieldRelated(
   812→                candidate.education[0]?.field,
   813→                job.required_education.field
   814→            );
   815→            if (isRelated) score += 5;
   816→        }
   817→    }
   818→    
   819→    // 4. Experience Match (10 points)
   820→    if (job.required_experience_years > 0) {
   821→        const expRatio = Math.min(
   822→            1.0,
   823→            cluster.years_experience / job.required_experience_years
   824→        );
   825→        score += expRatio * 10;
   826→    }
   827→    
   828→    return Math.round(score);
   829→}
   830→
   831→function getHighestEducationLevel(educationArray) {
   832→    const levels = ['None', 'Certificate', 'Diploma', 'Bachelor', 'Masters', 'PhD'];
   833→    let highest = 'None';
   834→    for (const edu of educationArray || []) {
   835→        const idx = levels.indexOf(edu.level);
   836→        const currentIdx = levels.indexOf(highest);
   837→        if (idx > currentIdx) {
   838→            highest = edu.level;
   839→        }
   840→    }
   841→    return highest;
   842→}
   843→```
   844→
   845→### Step 3: Education Level Comparison
   846→
   847→```javascript
   848→const EDUCATION_LEVELS = {
   849→    'PhD': 5,
   850→    'Masters': 4,
   851→    'Bachelor': 3,
   852→    'Diploma': 2,
   853→    'Certificate': 1,
   854→    'None': 0
   855→};
   856→
   857→function compareEducation(candidateLevel, jobRequiredLevel) {
   858→    const candidateValue = EDUCATION_LEVELS[candidateLevel] || 0;
   859→    const requiredValue = EDUCATION_LEVELS[jobRequiredLevel] || 0;
   860→    
   861→    if (candidateValue >= requiredValue) {
   862→        return 1.0;
   863→    } else if (candidateValue === requiredValue - 1) {
   864→        return 0.7;
   865→    } else if (candidateValue === requiredValue - 2) {
   866→        return 0.3;
   867→    } else {
   868→        return 0.0;
   869→    }
   870→}
   871→```
   872→
   873→### Step 4: Generate Explanation
   874→
   875→```javascript
   876→function generateExplanation(cluster, job) {
   877→    const matchedSkills = job.required_skills.filter(
   878→        skill => cluster.skills.some(s => s.toLowerCase() === skill.toLowerCase())
   879→    );
   880→    
   881→    const sampleTitle = cluster.job_titles[0] || 'your experience';
   882→    let explanation = `This job matches your experience as a ${sampleTitle}`;
   883→    
   884→    if (matchedSkills.length > 0) {
   885→        const skillList = matchedSkills.slice(0, 3).join(', ');
   886→        explanation += ` and your skills in ${skillList}`;
   887→        if (matchedSkills.length > 3) {
   888→            explanation += ` and ${matchedSkills.length - 3} other skills`;
   889→        }
   890→    }
   891→    
   892→    return explanation;
   893→}
   894→```
   895→
   896→### Step 5: Final Ranking with Scalable Query
   897→
   898→```javascript
   899→async function rankJobsForCandidate(candidateId, jobId) {
   900→    // Get candidate with their clusters
   901→    const candidate = await getCandidateWithClusters(candidateId);
   902→    const job = await getJob(jobId);
   903→    
   904→    // Only match if function exists
   905→    const activeClusters = candidate.work_experience.filter(c => c.selected);
   906→    let bestScore = 0;
   907→    let bestCluster = null;
   908→    
   909→    for (const cluster of activeClusters) {
   910→        if (cluster.function.toLowerCase() === job.job_function.toLowerCase()) {
   911→            const score = calculateMatchScore(cluster, job, candidate);
   912→            if (score > bestScore) {
   913→                bestScore = score;
   914→                bestCluster = cluster;
   915→            }
   916→        }
   917→    }
   918→    
   919→    if (bestCluster) {
   920→        return {
   921→            candidate_id: candidateId,
   922→            job_id: jobId,
   923→            match_score: bestScore,
   924→            matched_cluster_id: bestCluster.cluster_id,
   925→            explanation: generateExplanation(bestCluster, job)
   926→        };
   927→    }
   928→    
   929→    return null;
   930→}
   931→```

   933→## 6.3 What We Match On vs What We Don't
   934→
   935→### ✅ MATCH ON:
   936→
   937→| Factor | Weight | Purpose |
   938→|--------|--------|---------|
   939→| **Function** | Required | Primary filter - job must match a candidate cluster |
   940→| **Job Titles** | 40 points | Exact or partial title match |
   941→| **Skills** | 35 points | Percentage of required skills present |
   942→| **Education Level** | 15 points | Higher or equal to requirement |
   943→| **Education Field** | 5 points (bonus) | Related field |
   944→| **Experience Years** | 10 points | Years in matching cluster |
   945→
   946→### ❌ DO NOT MATCH ON:
   947→
   948→| Factor | Reason |
   949→|--------|--------|
   950→| **Driver's License** | Administrative requirement - candidate decides |
   951→| **Age** | Not relevant for matching |
   952→| **Nationality** | Not relevant for matching |
   953→| **Languages** | Not relevant for matching |
   954→| **Certifications** | Too varied, hard to standardize |
   955→| **Referees** | Not relevant for matching |
   956→| **Company Sector** | Too ambiguous to extract from CV |
   957→| **Grade/Class** | Not relevant for matching |
   958→| **Location** | Candidate decides if they want to relocate |

  1388→                    candidate_id, level, field, institution,

  1403→                    candidateId,

  1405→                    JSON.stringify(cluster.job_titles),

  1410→                ]

  1432→        // Trigger initial matching for this candidate against all jobs

  1437→            candidate_id: candidateId,

  1584→        let normalizedData = null;
  1585→        let rawJdText = null;
  1586→

  1588→            case 'paste':
  1589→                // Raw JD text → LLM extraction
  1590→                rawJdText = data;
  1591→                const llmResult = await parseJD(data);
  1592→                if (llmResult._error) {
  1593→                    return Response.json({ 
  1594→                        error: 'Failed to parse job description',
  1595→                        details: llmResult._message
  1596→                    }, { status: 400 });
  1597→                }
  1598→                normalizedData = normalizeJob(llmResult);
  1599→                break;
  1600→                
  1601→            case 'upload':
  1602→                // File upload → extract text → LLM
  1603→                const file = data.file;
  1604→                const extractedText = await extractText(file);
  1605→                rawJdText = extractedText;
  1606→                const llmResult2 = await parseJD(extractedText);
  1607→                if (llmResult2._error) {
  1608→                    return Response.json({ 
  1609→                        error: 'Failed to parse job description',
  1610→                        details: llmResult2._message
  1611→                    }, { status: 400 });
  1612→                }

  1614→                break;
  1615→                
  1616→            case 'json':
  1617→                // Already structured JSON → validate → use directly
  1618→                try {
  1619→                    const parsed = typeof data === 'string' ? JSON.parse(data) : data;
  1620→                    // Validate schema
  1621→                    validateJobSchema(parsed);
  1622→                    normalizedData = normalizeJob(parsed);
  1623→                } catch (e) {
  1624→                    return Response.json({ error: 'Invalid JSON format' }, { status: 400 });
  1625→                }
  1626→                break;
  1627→                
  1628→            case 'form':
  1629→                // Already structured from form fields
  1630→                normalizedData = normalizeJob({ job: data });

  1632→                
  1633→            default:
  1634→                return Response.json({ error: 'Invalid input type' }, { status: 400 });
  1635→        }
  1636→        
  1637→        // Store job
  1638→        const result = await query(
  1639→            `INSERT INTO jobs (
  1640→                title, company, sector, job_function,
  1641→                location, job_type, required_skills,
  1642→                required_education_level, required_education_field,
  1643→                required_experience_years, description,
  1644→                application_deadline, posted_by, raw_jd_text, parsed_data
  1645→            ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`,
  1646→            [
  1647→                normalizedData.job.title,
  1648→                normalizedData.job.company,
  1649→                normalizedData.job.sector,
  1650→                normalizedData.job.job_function,
  1651→                normalizedData.job.location,
  1652→                normalizedData.job.job_type,

  1655→                normalizedData.job.required_education?.field || 'Any',
  1656→                normalizedData.job.required_experience_years || 0,

  1658→                data.application_deadline || null,

  1662→            ]

  1893→### Matching Script with Scalable Query
  1894→
  1895→```typescript
  1896→// scripts/run-matching.ts
  1897→import { query } from '../lib/db';
  1898→import { rankJobsForCandidate } from '../lib/matching';
  1899→
  1900→async function runMatching() {
  1901→    console.log('🔄 Starting matching job...');
  1902→    
  1903→    // Get jobs posted in the last hour (incremental)
  1904→    const jobs = await query(
  1905→        `SELECT * FROM jobs 
  1906→         WHERE created_at > DATE_SUB(NOW(), INTERVAL 1 HOUR)
  1907→         ORDER BY created_at DESC`
  1908→    );
  1909→    
  1910→    if (jobs.length === 0) {
  1911→        console.log('No new jobs to match');
  1912→        return { processed: 0, matches: 0 };
  1913→    }
  1914→    
  1915→    console.log(`📋 Found ${jobs.length} new job(s) to match`);
  1916→    let totalMatches = 0;
  1917→    
  1918→    for (const job of jobs) {
  1919→        console.log(`  🔍 Matching job: ${job.title} (${job.job_function})`);
  1920→        
  1921→        // SCALABLE: Only get candidates with matching function
  1922→        const candidates = await query(
  1923→            `SELECT DISTINCT c.* 
  1924→             FROM candidates c
  1925→             JOIN candidate_clusters cc ON c.id = cc.candidate_id
  1926→             WHERE cc.function = ? 
  1927→               AND cc.selected = TRUE
  1928→             ORDER BY c.created_at DESC`,
  1929→            [job.job_function]
  1930→        );
  1931→        
  1932→        console.log(`    Found ${candidates.length} candidates with matching function`);
  1933→        
  1934→        for (const candidate of candidates) {
  1935→            // Get candidate clusters
  1936→            const clusters = await query(
  1937→                `SELECT * FROM candidate_clusters 
  1938→                 WHERE candidate_id = ? AND selected = TRUE`,
  1939→                [candidate.id]
  1940→            );
  1941→            
  1942→            const candidateWithClusters = {
  1943→                ...candidate,
  1944→                work_experience: clusters,
  1945→                education: await query(
  1946→                    `SELECT * FROM candidate_education 
  1947→                     WHERE candidate_id = ?`,
  1948→                    [candidate.id]
  1949→                )
  1950→            };
  1951→            
  1952→            // Calculate match
  1953→            const match = await rankJobsForCandidate(
  1954→                candidate.id,
  1955→                job.id,
  1956→                candidateWithClusters
  1957→            );
  1958→            
  1959→            if (match) {
  1960→                await query(
  1961→                    `INSERT INTO job_matches (
  1962→                        candidate_id, job_id, match_score,
  1963→                        matched_cluster_id, explanation
  1964→                    ) VALUES (?, ?, ?, ?, ?)
  1965→                    ON DUPLICATE KEY UPDATE
  1966→                        match_score = VALUES(match_score),
  1967→                        explanation = VALUES(explanation),
  1968→                        updated_at = CURRENT_TIMESTAMP`,
  1969→                    [
  1970→                        candidate.id,
  1971→                        job.id,
  1972→                        match.match_score,
  1973→                        match.matched_cluster_id,
  1974→                        match.explanation
  1975→                    ]
  1976→                );
  1977→                totalMatches++;
  1978→            }
  1979→        }
  1980→    }
  1981→    
  1982→    console.log(`✅ Matching completed: ${totalMatches} matches created/updated`);
  1983→    return { processed: jobs.length, matches: totalMatches };
  1984→}
  1985→
  1986→// Run if called directly
  1987→if (require.main === module) {
  1988→    runMatching()
  1989→        .then(() => process.exit(0))
  1990→        .catch((error) => {
  1991→            console.error('❌ Fatal error:', error);
  1992→            process.exit(1);
  1993→        });
  1994→}
  1995→
  1996→export { runMatching };
  1997→```
