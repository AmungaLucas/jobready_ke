    96→def make_table(headers, rows, col_widths=None):
    97→    """Create a styled table."""
    98→    available = W - 2 * MARGIN
    99→    if col_widths is None:
   100→        col_widths = [available / len(headers)] * len(headers)
   101→    else:
   102→        col_widths = [w * available for w in col_widths]
   103→
   104→    header_paras = [Paragraph(f'<b>{h}</b>', ParagraphStyle('TH', fontName='FreeSerif-Bold',
   105→        fontSize=9.5, leading=13, textColor=colors.white, alignment=TA_CENTER)) for h in headers]
   106→    data = [header_paras]
   107→    for row in rows:
   108→        data.append([Paragraph(str(c), ParagraphStyle('TD', fontName='FreeSerif',
   109→            fontSize=9, leading=13, textColor=TEXT_PRIMARY, alignment=TA_LEFT)) for c in row])
   110→
   111→    t = Table(data, colWidths=col_widths, repeatRows=1)
   112→    style_cmds = [
   113→        ('BACKGROUND', (0, 0), (-1, 0), HEADER_FILL),
   114→        ('TEXTCOLOR', (0, 0), (-1, 0), colors.white),
   115→        ('FONTNAME', (0, 0), (-1, 0), 'FreeSerif-Bold'),
   116→        ('FONTSIZE', (0, 0), (-1, 0), 9.5),
   117→        ('BOTTOMPADDING', (0, 0), (-1, 0), 8),
   118→        ('TOPPADDING', (0, 0), (-1, 0), 8),
   119→        ('ALIGN', (0, 0), (-1, 0), 'CENTER'),
   120→        ('GRID', (0, 0), (-1, -1), 0.5, BORDER),
   121→        ('VALIGN', (0, 0), (-1, -1), 'TOP'),
   122→        ('LEFTPADDING', (0, 0), (-1, -1), 6),
   123→        ('RIGHTPADDING', (0, 0), (-1, -1), 6),
   124→        ('TOPPADDING', (0, 1), (-1, -1), 5),
   125→        ('BOTTOMPADDING', (0, 1), (-1, -1), 5),
   126→    ]
   127→    for i in range(1, len(data)):
   128→        if i % 2 == 0:
   129→            style_cmds.append(('BACKGROUND', (0, i), (-1, i), TABLE_STRIPE))
   130→        else:
   131→            style_cmds.append(('BACKGROUND', (0, i), (-1, i), colors.white))
   132→    t.setStyle(TableStyle(style_cmds))
   133→    return t
